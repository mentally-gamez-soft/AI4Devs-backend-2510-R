# ATS API Test Queries - CURL Commands

## Base Configuration
BASE_URL="http://localhost:3010"

#############################################
# Position Endpoints - GET /positions/:id/candidates
#############################################

## Test 1: Get all candidates for a specific position (Success case)
curl -X GET "${BASE_URL}/positions/1/candidates" \
  -H "Content-Type: application/json"

## Test 2: Get candidates for non-existent position (404 case)
curl -X GET "${BASE_URL}/positions/99999/candidates" \
  -H "Content-Type: application/json"

## Test 3: Get candidates with invalid position ID format (400 case)
curl -X GET "${BASE_URL}/positions/invalid/candidates" \
  -H "Content-Type: application/json"

## Test 4: Get candidates for position 2
curl -X GET "${BASE_URL}/positions/2/candidates" \
  -H "Content-Type: application/json"

## Test 5: Get candidates with verbose output (see headers)
curl -v -X GET "${BASE_URL}/positions/1/candidates" \
  -H "Content-Type: application/json"

#############################################
# Candidate Stage Update - PUT /candidates/:id/stage
#############################################

# Workflow Tests
#############################################

## Test 15: Complete workflow - Get position candidates, update one candidate's stage
# Step 1: Get all candidates for position 1
curl -X GET "${BASE_URL}/positions/1/candidates" \
  -H "Content-Type: application/json" | jq '.'

# Step 2: Update the first candidate's stage to interview step 2
# (Replace candidate ID based on results from step 1)
curl -X PUT "${BASE_URL}/candidates/1/stage" \
  -H "Content-Type: application/json" \
  -d '{
    "interviewStepId": 2
  }' | jq '.'

# Step 3: Verify the update by getting position candidates again
curl -X GET "${BASE_URL}/positions/1/candidates" \
  -H "Content-Type: application/json" | jq '.'
